~~~sql
-- 人脸图片信息表
CREATE TABLE IF NOT EXISTS face_image (
    id INT AUTO_INCREMENT PRIMARY KEY COMMENT '主键',
    person_name VARCHAR(64) NOT NULL COMMENT '人员姓名',
    image_path VARCHAR(255) NOT NULL COMMENT '图片在本地磁盘的路径',
    feature BLOB COMMENT '人脸特征向量',
    feature_json JSON COMMENT '人脸特征向量（JSON格式）',
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间'
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='人脸图片信息表';

-- 识别日志表
CREATE TABLE IF NOT EXISTS face_recognition_log (
    id INT AUTO_INCREMENT PRIMARY KEY COMMENT '主键',
    image_id INT COMMENT '被识别的人脸图片ID',
    input_image_path VARCHAR(255) NOT NULL COMMENT '输入图片路径',
    result VARCHAR(16) NOT NULL COMMENT '识别结果（存在/不存在）',
    similarity FLOAT COMMENT '相似度',
    recognized_at DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '识别时间',
    FOREIGN KEY (image_id) REFERENCES face_image(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='人脸识别日志表'; 
~~~
